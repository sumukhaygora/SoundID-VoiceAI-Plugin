cmake_minimum_required(VERSION 3.22)
project(SoundIDVoiceAI VERSION 1.0.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 1. Fetch JUCE 8 automatically during build
include(FetchContent)
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG        8.0.3
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(JUCE)

# 2. Optional ONNX Runtime discovery
if(NOT DEFINED ONNXRUNTIME_ROOTDIR)
    set(ONNXRUNTIME_ROOTDIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime")
endif()

find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
    PATHS "${ONNXRUNTIME_ROOTDIR}/include" NO_DEFAULT_PATH)
find_library(ONNXRUNTIME_LIB onnxruntime
    PATHS "${ONNXRUNTIME_ROOTDIR}/lib" NO_DEFAULT_PATH)

# 3. Define Plugin & Standalone Targets (VST3 + AU + Standalone)
juce_add_plugin(SoundIDVoiceAI
    COMPANY_NAME                "SonarworksAI"
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    COPY_PLUGIN_AFTER_BUILD     FALSE
    PLUGIN_MANUFACTURER_CODE    SwAi
    PLUGIN_CODE                 SvAi
    FORMATS                     VST3 AU Standalone
    PRODUCT_NAME                "SoundID VoiceAI"
)

target_sources(SoundIDVoiceAI PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginProcessor.h
    Source/PluginEditor.cpp
    Source/PluginEditor.h
    Source/PluginParameters.h
    Source/UnisonRealtimeMixer.h
    Source/CaptureAndCacheManager.h
    Source/NeuralVoiceEngine.h
)

target_compile_definitions(SoundIDVoiceAI PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    $<$<BOOL:${ONNXRUNTIME_LIB}>:USE_ONNXRUNTIME=1>
)

if(ONNXRUNTIME_INCLUDE_DIR AND ONNXRUNTIME_LIB)
    target_include_directories(SoundIDVoiceAI PRIVATE ${ONNXRUNTIME_INCLUDE_DIR})
    target_link_libraries(SoundIDVoiceAI PRIVATE ${ONNXRUNTIME_LIB})
endif()

target_link_libraries(SoundIDVoiceAI
    PRIVATE
        juce::juce_audio_utils
        juce::juce_audio_processors
        juce::juce_dsp
        juce::juce_gui_extra
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)
